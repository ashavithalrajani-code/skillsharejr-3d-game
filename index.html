// ========================= // PLAYER // ========================= const player = new THREE.Group(); scene.add(player); // Player body const bodyGeometry = new THREE.CapsuleGeometry( 0.45, 1.2, 8, 16 ); const bodyMaterial = new THREE.MeshStandardMaterial({ color: 0x2468c9 }); const body = new THREE.Mesh( bodyGeometry, bodyMaterial ); body.position.y = 1.1; body.castShadow = true; player.add(body); // Player head const headGeometry = new THREE.SphereGeometry( 0.38, 16, 16 ); const headMaterial = new THREE.MeshStandardMaterial({ color: 0xffc49b }); const head = new THREE.Mesh( headGeometry, headMaterial ); head.position.y = 2.1; head.castShadow = true; player.add(head); // Player cloak const cloakGeometry = new THREE.ConeGeometry( 0.7, 1.5, 8 ); const cloakMaterial = new THREE.MeshStandardMaterial({ color: 0x172d68 }); const cloak = new THREE.Mesh( cloakGeometry, cloakMaterial ); cloak.rotation.x = Math.PI; cloak.position.set( 0, 1.1, 0.25 ); cloak.castShadow = true; player.add(cloak); // Player position player.position.set( 0, 0, 10 ); // ========================= // MOVEMENT // ========================= const keys = {}; window.addEventListener( "keydown", (event) => { keys[event.key.toLowerCase()] = true; } ); window.addEventListener( "keyup", (event) => { keys[event.key.toLowerCase()] = false; } ); // Movement speed const playerSpeed = 0.12; // ========================= // UPDATE PLAYER // ========================= function updatePlayer() { let moving = false; if (keys["w"]) { player.position.z -= playerSpeed; moving = true; } if (keys["s"]) { player.position.z += playerSpeed; moving = true; } if (keys["a"]) { player.position.x -= playerSpeed; moving = true; } if (keys["d"]) { player.position.x += playerSpeed; moving = true; } // Keep player inside the world player.position.x = THREE.MathUtils.clamp( player.position.x, -45, 45 ); player.position.z = THREE.MathUtils.clamp( player.position.z, -45, 45 ); // Small walking animation if (moving) { body.rotation.z = Math.sin( Date.now() * 0.015 ) * 0.08; } else { body.rotation.z = 0; } } // ========================= // CAMERA FOLLOW // ========================= function updateCamera() { const targetPosition = new THREE.Vector3( player.position.x, player.position.y + 7, player.position.z + 12 ); camera.position.lerp( targetPosition, 0.08 ); camera.lookAt( player.position.x, player.position.y + 1, player.position.z ); }
